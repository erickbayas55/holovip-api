from flask import Flask, jsonify
from flask_cors import CORS

app = Flask(__name__)
CORS(app)

@app.route('/')
def home():
    return jsonify({"message": "API HoloVip funcionando!", "status": "ok"})

@app.route('/api/test')
def test():
    return jsonify({"usuarios": 100, "activo": True})

if __name__ == '__main__':
    app.run()
